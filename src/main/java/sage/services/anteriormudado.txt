package sage.services;

import com.google.api.core.ApiFuture;
import com.google.cloud.Timestamp;
import com.google.cloud.firestore.*;
import com.google.firebase.cloud.FirestoreClient;
import jakarta.ws.rs.Consumes;
import jakarta.ws.rs.POST;
import jakarta.ws.rs.Path;
import jakarta.ws.rs.Produces;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import sage.models.*;

import java.time.Instant;
import java.time.LocalDate;
import java.time.LocalTime;
import java.time.ZoneId;
import java.util.*;
import java.util.concurrent.ExecutionException;
import java.util.stream.Collectors;

@Path("/reservas")
public class Reservas {

    Firestore db = FirestoreClient.getFirestore();

    @POST
    @Path("/salas-horarios")
    @Produces(MediaType.APPLICATION_JSON)
    public Response listarSalasComHorarios() {
        // Cria uma instância de AreasComHorariosResponse para acessar os locais
        AreasComHorariosResponse areasComHorarios = new AreasComHorariosResponse(null, null);

        // Obtém as áreas definidas no método getLocais
        List<Area> areas = areasComHorarios.getLocais();

        // Criando a resposta que contém as áreas e os horários comuns
        AreasComHorariosResponse response = new AreasComHorariosResponse(areas, Horarios.getHorarios());

        return Response.ok(response).build();
    }

    @POST
    @Path("/verificarReservas")
    @Produces(MediaType.APPLICATION_JSON)
    @Consumes(MediaType.APPLICATION_JSON)
    public Response verificarReservas(Map<String, String> requestData) {
        try {
            String eventoId = requestData.get("eventoId");

            // Referência à subcoleção "reservas" dentro do documento do evento
            CollectionReference reservasCollectionRef = db.collection("reservas").document(eventoId).collection("reservas");
            ApiFuture<QuerySnapshot> future = reservasCollectionRef.get();
            QuerySnapshot reservasSnapshot = future.get();

            if (reservasSnapshot.isEmpty()) {
                return Response.status(Response.Status.NOT_FOUND).entity("Reservas não encontradas para o evento: " + eventoId).build();
            }

            // Obtem a lista de documentos de reservas
            List<Map<String, Object>> reservasList = new ArrayList<>();
            for (DocumentSnapshot reservaDoc : reservasSnapshot.getDocuments()) {
                Map<String, Object> reservaData = reservaDoc.getData();
                if (reservaData != null) {
                    // Converte o campo "data" para Instant ISO string
                    reservaData.put("data", Instant.ofEpochMilli((Long) reservaData.get("data")).toString());
                    reservaData.put("id", reservaDoc.getId()); // Adiciona o ID da reserva
                    reservasList.add(reservaData);
                }
            }

            return Response.ok(reservasList).build();

        } catch (Exception e) {
            e.printStackTrace();
            return Response.status(Response.Status.INTERNAL_SERVER_ERROR)
                    .entity("Erro ao verificar as reservas: " + e.getMessage())
                    .build();
        }
    }


    @POST
    @Path("/reservasOcupadas")
    @Produces(MediaType.APPLICATION_JSON)
    @Consumes(MediaType.APPLICATION_JSON)
    public Response reservasOcupadas() {
        try {
            // Referência à coleção de reservas
            CollectionReference reservasCollection = db.collection("reservas");

            // Faz uma consulta para obter todos os documentos da coleção de reservas
            ApiFuture<QuerySnapshot> future = reservasCollection.get();
            List<QueryDocumentSnapshot> documents = future.get().getDocuments();

            List<Map<String, Object>> todasReservas = new ArrayList<>();

            // Verifica se a coleção tem documentos
            if (documents.isEmpty()) {
                System.out.println("Nenhum documento na coleção 'reservas'.");
                return Response.ok(Collections.emptyList()).build();
            }

            // Itera sobre todos os documentos da coleção de reservas
            for (QueryDocumentSnapshot document : documents) {
                System.out.println("Documento do evento: " + document.getId());

                // Obtem todas as reservas dentro da subcoleção "reservas" do evento atual
                CollectionReference reservasSubCollection = document.getReference().collection("reservas");
                ApiFuture<QuerySnapshot> reservasFuture = reservasSubCollection.get();
                List<QueryDocumentSnapshot> reservasDocuments = reservasFuture.get().getDocuments();

                // Verifica se a subcoleção tem documentos
                if (reservasDocuments.isEmpty()) {
                    System.out.println("Nenhuma reserva encontrada para o evento: " + document.getId());
                    continue; // Continua para o próximo documento de evento
                }

                // Itera sobre todas as reservas da subcoleção e coleta os dados
                for (QueryDocumentSnapshot reservaDoc : reservasDocuments) {
                    Map<String, Object> reservaData = reservaDoc.getData();

                    // Verifica se o campo "data" está presente
                    if (reservaData.get("data") != null) {
                        Long timestamp = (Long) reservaData.get("data");
                        reservaData.put("data", Instant.ofEpochMilli(timestamp).toString());
                    } else {
                        System.out.println("Reserva sem campo 'data': " + reservaDoc.getId());
                    }

                    // Log detalhado da reserva
                    System.out.println("Reserva encontrada: " + reservaDoc.getId() + " com dados: " + reservaData);

                    // Adiciona a reserva à lista principal
                    todasReservas.add(reservaData);
                }
            }

            // Verifica se alguma reserva foi adicionada
            if (todasReservas.isEmpty()) {
                System.out.println("Nenhuma reserva ocupada encontrada.");
                return Response.ok(Collections.emptyList()).build();
            }

            // Retorna todas as reservas ocupadas (com dados de sala, horários e data)
            return Response.ok(todasReservas).build();

        } catch (Exception e) {
            e.printStackTrace();
            return Response.status(Response.Status.INTERNAL_SERVER_ERROR)
                    .entity("Erro ao verificar as reservas ocupadas: " + e.getMessage())
                    .build();
        }
    }





    @POST
    @Path("/excluirReserva")
    @Produces(MediaType.APPLICATION_JSON)
    @Consumes(MediaType.APPLICATION_JSON)
    public Response excluirReserva(Map<String, Object> dados) {
        try {
            String eventoId = (String) dados.get("eventoId");
            Integer index = (Integer) dados.get("index");

            if (eventoId == null || index == null) {
                return Response.status(Response.Status.BAD_REQUEST)
                        .entity("Evento ID ou index inválido.")
                        .build();
            }

            // Referência ao Firestore
            Firestore db = FirestoreClient.getFirestore();

            // Acessa o documento de reservas associado ao eventoId
            DocumentReference eventoDocRef = db.collection("reservas").document(eventoId);
            ApiFuture<DocumentSnapshot> future = eventoDocRef.get();
            DocumentSnapshot document = future.get();

            if (!document.exists()) {
                return Response.status(Response.Status.NOT_FOUND)
                        .entity("Evento não encontrado.")
                        .build();
            }

            // Obtém as reservas como um List<Map<String, Object>>
            List<Map<String, Object>> reservas = (List<Map<String, Object>>) document.get("reservas");

            if (reservas != null && index < reservas.size()) {
                // Remove a reserva no índice especificado
                reservas.remove((int) index);

                // Atualiza o documento com a lista modificada de reservas
                ApiFuture<WriteResult> writeResult = eventoDocRef.update("reservas", reservas);
                writeResult.get();  // Aguarda a conclusão da escrita

                return Response.ok("Reserva excluída com sucesso").build();
            } else {
                return Response.status(Response.Status.BAD_REQUEST)
                        .entity("Index inválido ou lista de reservas vazia.")
                        .build();
            }
        } catch (Exception e) {
            e.printStackTrace();
            return Response.status(Response.Status.INTERNAL_SERVER_ERROR)
                    .entity("Erro ao excluir a reserva: " + e.getMessage())
                    .build();
        }
    }

    @POST
    @Path("/reservar")
    @Produces(MediaType.APPLICATION_JSON)
    @Consumes(MediaType.APPLICATION_JSON)
    public Response reservar(ReservasRequest reservasRequest) {
        try {
            String eventoId = reservasRequest.getEventoId();
            List<ReservarSalas> novasReservas = reservasRequest.getReservas();

            // Refere-se ao documento do evento
            DocumentReference eventoDocRef = db.collection("reservas").document(eventoId);

            for (ReservarSalas novaReserva : novasReservas) {
                boolean reservaExistente = false;

                // Converte o timestamp da nova reserva para LocalDate
                LocalDate novaReservaDia = Instant.ofEpochMilli(novaReserva.getData())
                        .atZone(ZoneId.systemDefault()).toLocalDate();

                // Busca todas as reservas existentes na subcoleção "reservas"
                ApiFuture<QuerySnapshot> futureReservas = eventoDocRef.collection("reservas").get();
                List<QueryDocumentSnapshot> reservasExistentes = futureReservas.get().getDocuments();

                for (QueryDocumentSnapshot reservaAtual : reservasExistentes) {
                    // Converte o timestamp da reserva atual para LocalDate
                    LocalDate reservaAtualDia = Instant.ofEpochMilli((Long) reservaAtual.get("data"))
                            .atZone(ZoneId.systemDefault()).toLocalDate();

                    // Verifica se a sala e a data (apenas dia) são iguais
                    if (reservaAtual.get("sala").equals(novaReserva.getSala()) &&
                            reservaAtualDia.equals(novaReservaDia)) {

                        // Adiciona os novos horários à reserva existente sem duplicar
                        List<String> horariosExistentes = (List<String>) reservaAtual.get("horarios");
                        List<String> novosHorarios = novaReserva.getHorarios();

                        // Adiciona novos horários sem duplicar
                        for (String novoHorario : novosHorarios) {
                            if (!horariosExistentes.contains(novoHorario)) {
                                horariosExistentes.add(novoHorario);
                            }
                        }

                        // Ordena os horários existentes
                        Collections.sort(horariosExistentes, Comparator.comparing(horario -> LocalTime.parse(horario.split(" - ")[0])));

                        // Atualiza a reserva existente
                        reservaAtual.getReference().update("horarios", horariosExistentes);

                        reservaExistente = true;
                        break; // Encerra o loop ao encontrar a reserva correspondente
                    }
                }

                if (!reservaExistente) {
                    // Cria uma nova reserva se não houver nenhuma para a mesma sala e data
                    DocumentReference novaReservaRef = eventoDocRef.collection("reservas").document();

                    Map<String, Object> novaReservaData = new HashMap<>();
                    novaReservaData.put("area", novaReserva.getArea());
                    novaReservaData.put("sala", novaReserva.getSala());
                    novaReservaData.put("data", novaReserva.getData()); // Mantém o timestamp completo
                    novaReservaData.put("horarios", novaReserva.getHorarios());

                    // Salva a nova reserva na subcoleção
                    novaReservaRef.set(novaReservaData);
                }
            }

            return Response.ok("Reservas atualizadas com sucesso.").build();

        } catch (Exception e) {
            return Response.status(Response.Status.INTERNAL_SERVER_ERROR)
                    .entity("Erro ao processar reservas: " + e.getMessage())
                    .build();
        }
    }




}
